# Prefill Fabric examples.

add_executable(basic_cpu_prefill basic_cpu_prefill.cpp)
target_link_libraries(basic_cpu_prefill PRIVATE PrefillFabric)
if(MSVC)
  target_compile_options(basic_cpu_prefill PRIVATE /W4 /WX)
endif()
set_target_properties(basic_cpu_prefill PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/examples)
